if (!sheetSoal) { sheetSoal = ss.insertSheet(SHEET_DATA_SOAL); // Set kolom contoh Data Soal sheetSoal.appendRow(["Nomor Soal", "Soal", "Pilihan A", "Pilihan B", "Pilihan C", "Pilihan D", "Jawaban Benar"]); } let sheetHasil = ss.getSheetByName(SHEET_HASIL); if (!sheetHasil) { sheetHasil = ss.insertSheet(SHEET_HASIL); // Kolom dasar hasil kuis // Detail inputan jawaban siswa dimasukkan dalam satu kolom JSON/string sheetHasil.appendRow([ "Timestamp", "Nama Siswa", "Kelas", "Jumlah Jawaban Benar", "Jumlah Jawaban Salah", "Detail Jawaban Siswa" ]); } } /** * Entry point Web App POST untuk menerima data dari frontend * Menyimpan hasil LKPD dan Kuis ke Sheet "Hasil" */ function doPost(e) { // Parsing data dari request POST JSON let ss = SpreadsheetApp.getActiveSpreadsheet(); } // Hitung jumlah benar & salah jika tersedia jawaban benar (opsional) // Jika data soal diperlukan ada di sheet, tapi di sini backend cuma menyimpan hasil let benar = 0; let salah = 0; if (data.jawabanBenar && typeof data.jawabanBenar === 'number' && data.jawabanSalah && typeof data.jawabanSalah === 'number') { benar = data.jawabanBenar; salah = data.jawabanSalah; } // Simpan sebagai baris baru sheetHasil.appendRow([ new Date(), data.nama, data.kelas, benar, salah, JSON.stringify(data.detailJawaban) ]); return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Data tersimpan"})) .setMimeType(ContentService.MimeType.JSON); }); } /** * Optional GET untuk debugging/load soal */ function doGet(e) { let ss = SpreadsheetApp.getActiveSpreadsheet(); setupSheets(ss); let sheetSoal = ss.getSheetByName(SHEET_DATA_SOAL); let data = sheetSoal.getDataRange().getValues(); C. Panduan Deployment 1. Menyiapkan Google Spreadsheet & Apps Script Web App Buat Google Spreadsheet baru di Google Drive Anda. Klik menu Extensions > Apps Script untuk membuka editor Apps Script. Di editor Apps Script: Hapus kode yang ada dan tempelkan kode Code.gs yang saya berikan di atas. Simpan project dengan nama misalnya "Backend MPI Pecahan". Deploy sebagai Web App: Klik Deploy > New Deployment. Pilih Type: Web app. Beri nama versi (contoh: "v1"). Pada "Execute as", pilih Me (pemilik script). Pada "Who has access", pilih Anyone (open access, karena dipanggil dari GitHub Pages). Klik Deploy. Beri otorisasi yang diperlukan (izinkan akses). Setelah deploy selesai, salin URL Web App yang diberikan, ini adalah URL yang harus digunakan untuk GAS_URL di frontend HTML Anda. 2. Men-Deploy Frontend ke GitHub Pages Buat repositori GitHub baru, misalnya dengan nama mpi-pecahan. Upload file HTML Anda (misal file index.html yang sudah dimodifikasi) ke repositori. Di GitHub repository: Pergi ke Settings > Pages. Pada "Source" pilih branch (biasanya main) dan folder root (/root) — lalu klik Save. GitHub akan membangun halaman dan memberikan URL halaman Anda, misal https://username.github.io/mpi-pecahan/. Buka URL tersebut dan lakukan pengujian aplikasi. Dengan setup ini, data jawaban siswa dari frontend akan tersimpan secara otomatis ke Google Sheets "Hasil", dan jika perlu Anda bisa mengelola soal di sheet "Data Soal".
